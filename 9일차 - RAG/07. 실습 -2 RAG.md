# =====================================================
# 1. 라이브러리 Import 및 API Key 설정
# =====================================================
import os
import wikipedia

from google.colab import userdata
from llama_index.core import VectorStoreIndex
from llama_index.readers.wikipedia import WikipediaReader

os.environ["OPENAI_API_KEY"] = userdata.get("OPENAI_API_KEY")


# =====================================================
# 2. Wikipedia 검색 대상 설정
# =====================================================
city_names = ["Seoul", "Berlin", "Dresden", "Suwon"]
# =====================================================
# 3. Wikipedia → Document 생성
# =====================================================
wikipedia.set_user_agent("my-rag-project (test@example.com)")

reader = WikipediaReader()

documents = reader.load_data(
    city_names,
    auto_suggest=False
)


# =====================================================
# 4. Vector Index 생성
# =====================================================
index = VectorStoreIndex.from_documents(documents)
# =====================================================
# 5. Query Engine 생성
# =====================================================
query_engine = index.as_query_engine()
# =====================================================
# 6. RAG 질문 실행
# =====================================================
question = "Which Korean city has a relationship with Dresden?"
response = query_engine.query(question)
print(response)


# =====================================================
# 7. 실제 검색된 Node 확인
# =====================================================
retriever = index.as_retriever()
nodes = retriever.retrieve(question)

for node in nodes:
    print(node.text)

# Custom RAG -Retreive와 Generate 연결
```python
# =====================================================
# 1. 필요한 클래스 Import
# =====================================================
from llama_index.core.query_engine import CustomQueryEngine
from llama_index.core.retrievers import BaseRetriever
from llama_index.core import PromptTemplate
from llama_index.llms.openai import OpenAI


# =====================================================
# 2. LLM에 전달할 Prompt 정의
# =====================================================
simple_qa_prompt = PromptTemplate(
    "Context information is below.\n"
    "---------------------\n"
    "{context_str}\n"
    "---------------------\n"
    "Given the context information and not prior knowledge, "
    "answer the query.\n"
    "Query: {query_str}\n"
    "Answer: "
)
# =====================================================
# 3. Custom RAG Query Engine 정의
# =====================================================
class OurCustomQueryEngine(CustomQueryEngine):

    retriever: BaseRetriever
    llm: OpenAI
    qa_prompt: PromptTemplate = simple_qa_prompt

    def custom_query(self, query_str: str):
        # ① 질문과 관련된 Node 검색
        nodes = self.retriever.retrieve(query_str)
        # ② 검색된 Node들을 하나의 Context로 결합
        context_str = "\n\n".join(
            [n.node.get_content() for n in nodes]
        )
        # ③ Context + 질문을 LLM에 전달
        response = self.llm.complete(
            self.qa_prompt.format(
                context_str=context_str,
                query_str=query_str
            )
        )
        # ④ 답변 반환
        return str(response)


# =====================================================
# 4. Retriever / LLM 생성
# =====================================================
retriever = index.as_retriever()
llm = OpenAI(model="gpt-3.5-turbo")
# =====================================================
# 5. Custom Query Engine 생성
# =====================================================
query_engine_answer = OurCustomQueryEngine(
    retriever=retriever,
    llm=llm,
    qa_prompt=simple_qa_prompt
)
# =====================================================
# 6. RAG 질문 실행
# =====================================================
response = query_engine_answer.query(
    "What's the arts and culture scene in Berlin?"
)

print(response)

```
